"use client";

import React, { useEffect, useRef, useState } from "react";

// FaceMesh class (runtime)
import { FaceMesh } from "@mediapipe/face_mesh";

// Landmark connection groups — note the explicit path and .js
import {
  FACEMESH_LIPS,
  FACEMESH_LEFT_EYE,
  FACEMESH_RIGHT_EYE,
  FACEMESH_LEFT_EYEBROW,
  FACEMESH_RIGHT_EYEBROW,
  FACEMESH_FACE_OVAL
} from "@mediapipe/face_mesh/face_mesh_connections.js";


// Segmentation
import { SelfieSegmentation } from "@mediapipe/selfie_segmentation";

// Utils for camera capture
//import { Camera as MpCamera } from "@mediapipe/camera_utils";

import { Camera, X } from "lucide-react";

/* -------------------------------------------------------
   Helper Functions (JS version)
------------------------------------------------------- */

function lmToPoint(lm, width, height) {
  return { x: lm.x * width, y: lm.y * height };
}

function polygonFromConnections(landmarks, connections, width, height) {
  const indices = new Set();

  connections.forEach(([a, b]) => {
    indices.add(a);
    indices.add(b);
  });

  const points = Array.from(indices).map((i) =>
    lmToPoint(landmarks[i], width, height)
  );

  if (!points.length) return [];

  const cx = points.reduce((s, p) => s + p.x, 0) / points.length;
  const cy = points.reduce((s, p) => s + p.y, 0) / points.length;

  return points.sort(
    (p1, p2) =>
      Math.atan2(p1.y - cy, p1.x - cx) - Math.atan2(p2.y - cy, p2.x - cx)
  );
}

function expandPolygon(poly, scale = 1.3) {
  if (!poly || !poly.length) return poly;

  const cx = poly.reduce((s, p) => s + p.x, 0) / poly.length;
  const cy = poly.reduce((s, p) => s + p.y, 0) / poly.length;

  return poly.map((p) => ({
    x: cx + (p.x - cx) * scale,
    y: cy + (p.y - cy) * scale,
  }));
}

function clipToPolygon(ctx, poly) {
  ctx.beginPath();
  poly.forEach((pt, i) =>
    i === 0 ? ctx.moveTo(pt.x, pt.y) : ctx.lineTo(pt.x, pt.y)
  );
  ctx.closePath();
  ctx.clip();
}

/* -------------------------------------------------------
   Camera Modal (JSX)
------------------------------------------------------- */
function CameraModal({ onCapture, onClose }) {
  const videoRef = useRef(null);
  const snapCanvasRef = useRef(null);

  useEffect(() => {
    let mounted = true;
    navigator.mediaDevices
      .getUserMedia({ video: { facingMode: "user" } })
      .then((stream) => {
        if (mounted && videoRef.current) {
          videoRef.current.srcObject = stream;
        }
      });
    return () => {
      mounted = false;
      if (videoRef.current && videoRef.current.srcObject) {
        const s = videoRef.current.srcObject;
        s.getTracks().forEach((t) => t.stop());
      }
    };
  }, []);

  const handleSnap = () => {
    const video = videoRef.current;
    const canvas = snapCanvasRef.current;

    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;

    const ctx = canvas.getContext("2d");
    ctx.drawImage(video, 0, 0);

    canvas.toBlob((blob) => {
      const file = new File([blob], "camera.jpg", { type: "image/jpeg" });
      onCapture(file);
      onClose();
    }, "image/jpeg");
  };

  return (
    <div className="fixed inset-0 bg-black/60 flex items-center justify-center z-50">
      <div className="bg-white rounded-xl p-4 w-full max-w-md relative">
        <button className="absolute right-3 top-3" onClick={onClose}>
          <X />
        </button>

        <video
          ref={videoRef}
          autoPlay
          playsInline
          muted
          className="w-full rounded-md"
        />

        <canvas ref={snapCanvasRef} className="hidden" />

        <button
          onClick={handleSnap}
          className="w-full py-2 bg-black text-white rounded mt-3"
        >
          Capture
        </button>
      </div>
    </div>
  );
}

/* -------------------------------------------------------
   MAIN COMPONENT — JSX VERSION
------------------------------------------------------- */

export default function MediaPipePictureMakeup() {
  const [preview, setPreview] = useState(null);
  const [modelsReady, setModelsReady] = useState(false);
  const [showCamera, setShowCamera] = useState(false);

  const fileRef = useRef(null);
  const imgRef = useRef(null);
  const canvasRef = useRef(null);
  const segCanvasRef = useRef(null);

  const faceMeshRef = useRef(null);
  const segRef = useRef(null);

  const polygonsRef = useRef({});

  /* Makeup controls */
  const [lipColor, setLipColor] = useState("#c81d3f");
  const [lipOpacity, setLipOpacity] = useState(0.85);

  const [eyeColor, setEyeColor] = useState("#6b2fa3");
  const [eyeOpacity, setEyeOpacity] = useState(0.45);

  const [linerColor, setLinerColor] = useState("#000000");
  const [linerWidth, setLinerWidth] = useState(3);

  const [blushColor, setBlushColor] = useState("#ff6f91");
  const [blushOpacity, setBlushOpacity] = useState(0.25);

  const [skinTint, setSkinTint] = useState("#f3c5a2");
  const [skinSmooth, setSkinSmooth] = useState(0.22);

  /* -------------------------------------------------------
     Initialize MediaPipe FaceMesh + Segmentation
  ------------------------------------------------------- */
  useEffect(() => {
    let mounted = true;

    const fm = new FaceMesh({
      locateFile: (file) =>
        `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}`,
    });

    fm.setOptions({
      maxNumFaces: 1,
      refineLandmarks: true,
      minDetectionConfidence: 0.5,
      minTrackingConfidence: 0.5,
    });

    fm.onResults((results) => {
      const lmSet = results.multiFaceLandmarks && results.multiFaceLandmarks[0];
      if (!lmSet || !canvasRef.current) return;

      const canvas = canvasRef.current;
      const w = canvas.width;
      const h = canvas.height;

      const mouth = polygonFromConnections(lmSet, FACEMESH_LIPS, w, h);
      const leftEye = polygonFromConnections(lmSet, FACEMESH_LEFT_EYE, w, h);
      const rightEye = polygonFromConnections(lmSet, FACEMESH_RIGHT_EYE, w, h);
      const leftBrow = polygonFromConnections(
        lmSet,
        FACEMESH_LEFT_EYEBROW,
        w,
        h
      );
      const rightBrow = polygonFromConnections(
        lmSet,
        FACEMESH_RIGHT_EYEBROW,
        w,
        h
      );
      const faceOval = polygonFromConnections(lmSet, FACEMESH_FACE_OVAL, w, h);

      // Expand for coverage
      polygonsRef.current = {
        mouth: expandPolygon(mouth, 1.35),
        leftEye: expandPolygon(leftEye, 1.4),
        rightEye: expandPolygon(rightEye, 1.4),
        leftBrow,
        rightBrow,
        faceOval,
      };

      // Generate cheek approximations from oval
      if (faceOval.length > 10) {
        polygonsRef.current.leftCheek = expandPolygon(
          faceOval.slice(2, 7),
          1.2
        );
        polygonsRef.current.rightCheek = expandPolygon(
          faceOval.slice(9, 15),
          1.2
        );
      }

      renderMakeup();
    });

    faceMeshRef.current = fm;

    const ss = new SelfieSegmentation({
      locateFile: (file) =>
        `https://cdn.jsdelivr.net/npm/@mediapipe/selfie_segmentation/${file}`,
    });

    ss.setOptions({ modelSelection: 1 });

    ss.onResults((res) => {
      if (!segCanvasRef.current || !canvasRef.current) return;

      const segC = segCanvasRef.current;
      segC.width = canvasRef.current.width;
      segC.height = canvasRef.current.height;

      const sctx = segC.getContext("2d");
      sctx.clearRect(0, 0, segC.width, segC.height);
      sctx.drawImage(res.segmentationMask, 0, 0, segC.width, segC.height);

      renderMakeup();
    });

    segRef.current = ss;

    if (mounted) setModelsReady(true);

    return () => {
      mounted = false;
    };
  }, []);

  /* -------------------------------------------------------
     Load Image + Run Models
  ------------------------------------------------------- */
  useEffect(() => {
    if (!preview || !modelsReady) return;

    const img = new Image();
    imgRef.current = img;

    img.onload = async () => {
      const canvas = canvasRef.current;
      canvas.width = img.naturalWidth;
      canvas.height = img.naturalHeight;

      const ctx = canvas.getContext("2d");
      ctx.drawImage(img, 0, 0);

      if (segRef.current) await segRef.current.send({ image: img });
      if (faceMeshRef.current) await faceMeshRef.current.send({ image: img });

      renderMakeup();
    };

    img.src = preview;
  }, [preview, modelsReady]);

  /* -------------------------------------------------------
     Core Render Makeup
  ------------------------------------------------------- */
  function renderMakeup() {
    if (!canvasRef.current || !imgRef.current) return;

    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");

    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(imgRef.current, 0, 0);

    // Skin smoothing
    if (skinSmooth > 0 && segCanvasRef.current) {
      const blurred = document.createElement("canvas");
      blurred.width = canvas.width;
      blurred.height = canvas.height;

      const bctx = blurred.getContext("2d");
      bctx.drawImage(canvas, 0, 0);

      bctx.filter = `blur(${Math.round(skinSmooth * 24)}px)`;
      bctx.drawImage(blurred, 0, 0);
      bctx.filter = "none";

      const temp = document.createElement("canvas");
      temp.width = canvas.width;
      temp.height = canvas.height;

      const tctx = temp.getContext("2d");
      tctx.drawImage(blurred, 0, 0);
      tctx.globalCompositeOperation = "destination-in";
      tctx.drawImage(segCanvasRef.current, 0, 0);
      tctx.globalCompositeOperation = "source-over";

      ctx.drawImage(temp, 0, 0);
    }

    // Foundation tint
    ctx.save();
    ctx.globalAlpha = 0.22;
    ctx.globalCompositeOperation = "overlay";
    ctx.fillStyle = skinTint;
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.restore();

    // Lips
    const mouth = polygonsRef.current.mouth;
    if (mouth && lipOpacity > 0) {
      ctx.save();
      clipToPolygon(ctx, mouth);
      ctx.globalCompositeOperation = "soft-light";
      ctx.globalAlpha = lipOpacity;
      ctx.fillStyle = lipColor;
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.restore();
    }

    // Eyeshadow
    ["leftEye", "rightEye"].forEach((side) => {
      const poly = polygonsRef.current[side];
      if (poly) {
        ctx.save();
        clipToPolygon(ctx, poly);
        ctx.globalCompositeOperation = "soft-light";
        ctx.globalAlpha = eyeOpacity;
        ctx.fillStyle = eyeColor;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.restore();
      }
    });

    // Eyeliner
    const drawLine = (points) => {
      if (!points || !points.length) return;
      ctx.beginPath();
      points.forEach((p, i) =>
        i === 0 ? ctx.moveTo(p.x, p.y) : ctx.lineTo(p.x, p.y)
      );
      ctx.strokeStyle = linerColor;
      ctx.lineWidth = linerWidth;
      ctx.lineCap = "round";
      ctx.lineJoin = "round";
      ctx.stroke();
    };

    drawLine(polygonsRef.current.leftBrow);
    drawLine(polygonsRef.current.rightBrow);

    // Blush
    ["leftCheek", "rightCheek"].forEach((side) => {
      const poly = polygonsRef.current[side];
      if (poly) {
        ctx.save();
        clipToPolygon(ctx, poly);
        ctx.globalCompositeOperation = "soft-light";
        ctx.globalAlpha = blushOpacity;
        ctx.fillStyle = blushColor;
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.restore();
      }
    });
  }

  /* -------------------------------------------------------
     UI
  ------------------------------------------------------- */
  return (
    <div className="max-w-3xl mx-auto p-4 space-y-4">
      {showCamera && (
        <CameraModal
          onCapture={(f) => {
            const url = URL.createObjectURL(f);
            setPreview(url);
          }}
          onClose={() => setShowCamera(false)}
        />
      )}

      <div className="flex gap-3 items-center">
        <input
          type="file"
          ref={fileRef}
          className="hidden"
          accept="image/*"
          onChange={(e) =>
            e.target.files?.[0] &&
            setPreview(URL.createObjectURL(e.target.files[0]))
          }
        />

        <button
          onClick={() => fileRef.current.click()}
          className="px-4 py-2 bg-black text-white rounded"
        >
          Upload Image
        </button>

        <button
          onClick={() => setShowCamera(true)}
          className="px-4 py-2 bg-pink-600 text-white rounded flex items-center gap-2"
        >
          <Camera size={16} /> Use Camera
        </button>

        <div className="ml-auto text-sm text-slate-500">
          {modelsReady ? "Models ready" : "Loading models..."}
        </div>
      </div>

      <div className="rounded border overflow-hidden">
        <canvas ref={canvasRef} style={{ width: "100%", display: "block" }} />
      </div>

      <canvas ref={segCanvasRef} className="hidden" />

      {/* Makeup Controls */}
      <div className="grid grid-cols-2 gap-4">
        <div>
          <label className="block text-xs">Lip Color</label>
          <input
            type="color"
            value={lipColor}
            onChange={(e) => setLipColor(e.target.value)}
          />
          <input
            type="range"
            min="0"
            max="1"
            step="0.01"
            value={lipOpacity}
            onChange={(e) => setLipOpacity(e.target.value)}
          />
        </div>

        <div>
          <label className="block text-xs">Eyeshadow</label>
          <input
            type="color"
            value={eyeColor}
            onChange={(e) => setEyeColor(e.target.value)}
          />
          <input
            type="range"
            min="0"
            max="1"
            step="0.01"
            value={eyeOpacity}
            onChange={(e) => setEyeOpacity(e.target.value)}
          />
        </div>

        <div>
          <label className="block text-xs">Eyeliner</label>
          <input
            type="color"
            value={linerColor}
            onChange={(e) => setLinerColor(e.target.value)}
          />
          <input
            type="range"
            min="1"
            max="8"
            step="1"
            value={linerWidth}
            onChange={(e) => setLinerWidth(e.target.value)}
          />
        </div>

        <div>
          <label className="block text-xs">Blush</label>
          <input
            type="color"
            value={blushColor}
            onChange={(e) => setBlushColor(e.target.value)}
          />
          <input
            type="range"
            min="0"
            max="1"
            step="0.01"
            value={blushOpacity}
            onChange={(e) => setBlushOpacity(e.target.value)}
          />
        </div>

        <div className="col-span-2">
          <label className="block text-xs">Skin Tint</label>
          <input
            type="color"
            value={skinTint}
            onChange={(e) => setSkinTint(e.target.value)}
          />

          <label className="block text-xs mt-2">Skin Smoothing</label>
          <input
            type="range"
            min="0"
            max="1"
            step="0.01"
            value={skinSmooth}
            onChange={(e) => setSkinSmooth(e.target.value)}
          />
        </div>
      </div>
    </div>
  );
}
